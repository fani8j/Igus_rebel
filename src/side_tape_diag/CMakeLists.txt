cmake_minimum_required(VERSION 3.8)
project(side_tape_diag)
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra)
endif()
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
add_executable(servo_proximity_probe src/servo_proximity_probe.cpp)
ament_target_dependencies(servo_proximity_probe rclcpp moveit_core moveit_ros_planning)
install(TARGETS servo_proximity_probe DESTINATION lib/${PROJECT_NAME})
ament_package()
